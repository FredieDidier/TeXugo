\question Show that, if $f(x) = x^2$, then $f'(x) = 2x$ from the definition of the derivative.
\begin{solution}
  $f'(x) = \lim_{h\to 0} \frac{(x+h)^2 - x^2}{h} = \lim_{h \to 0} (2x + h) = 2x$.
\end{solution}

\question Solve the system.

  \begin{solution}$x = 1$, $y = 2$.\end{solution}
\end{parts}
