\question Mostre que, se $f(x) = x^2$, então $f'(x) = 2x$ pela definição de derivada.
\begin{solution}
  $f'(x) = \lim_{h\to 0} \frac{(x+h)^2 - x^2}{h} = \lim_{h \to 0} (2x + h) = 2x$.
\end{solution}

\question Resolva o sistema.
\begin{parts}
  \part $x + y = 5$ e $x - y = 1$.
  \begin{solution}$x = 3$, $y = 2$.\end{solution}
  \part $2x + y = 4$ e $x + 3y = 7$.
  \begin{solution}$x = 1$, $y = 2$.\end{solution}
\end{parts}
