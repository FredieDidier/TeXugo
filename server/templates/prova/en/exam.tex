\question[2] Define opportunity cost and give an example.
\begin{solution}
  The value of the best alternative given up when making a choice.
\end{solution}
\vspace{4cm}

\question Consider the demand $Q = 100 - 2P$.
\begin{parts}
  \part[1] Compute the quantity demanded when $P = 10$.
  \begin{solution}$Q = 80$.\end{solution}
  \vspace{2.5cm}
  \part[2] Compute the price elasticity at that point.
  \begin{solution}$\varepsilon = -2 \cdot 10/80 = -0.25$.\end{solution}
  \vspace{2.5cm}
\end{parts}

\question[1] Choose the correct option: a Giffen good is one whose demand
\begin{checkboxes}
  \choice falls when income rises.
  \CorrectChoice rises when its own price rises.
  \choice does not depend on the price.
  \choice rises when the price of a substitute falls.
\end{checkboxes}

\question[2] True or false? Explain: ``Every inferior good is a Giffen good.''
\begin{solution}
  False. Every Giffen good is inferior, but not the other way around.
\end{solution}
